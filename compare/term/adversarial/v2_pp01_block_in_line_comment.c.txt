// expect: terminates
// A /* inside a // comment starts nothing (C11 6.4.9): x = 0 runs and the loop is skipped. Removing /* ... */ before
// // comments swallows x = 0 up to the next */.
int main(void) {
  int x = 1;
  // a note about /* comments
  x = 0;
  /* reset done */
  while (x == 1) { }
  return 0;
}
