// expect: nonterminating. f writes the global g on its way back; after f(3), g is 1 and the loop runs forever. A call
// read as leaving the globals as they were would keep g at 0.
int g = 0;
void f(int n) {
  if (n > 0) {
    f(n - 1);
  }
  g = 1;
}
int main() {
  f(3);
  while (g == 1) {
  }
  return 0;
}
